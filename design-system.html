Standard chart options snippet

const THEME = {
  text:  '#e8eef8',
  muted: '#a8bdd6',
  grid:  '#213352',
  pA:    '#60A8D6',
  pB:    '#3CC8B8',
  pC:    '#FF9040',
  rd:    '#E86060',
  t3:    '#607a96',
};

// Shared scale defaults
scales: {
  x: {
    ticks: { color: THEME.muted },
    grid:  { color: THEME.grid },
    title: { color: THEME.muted }
  },
  y: {
    ticks: { color: THEME.muted },
    grid:  { color: THEME.grid }
  }
}

// Legend
legend: {
  labels: {
    color: THEME.muted,
    font:  { size: 11 },
    boxWidth: 10,
    padding: 8
  }
}

Header Rules
  • Sticky at top:0; z-index:30 with backdrop-filter:blur(12px)
  • Background rgba(11,16,24,0.92) — translucent so page bg gradient bleeds through at edges
  • Border bottom 1px solid var(--b1)
  • Left — brand name + live dot pill + date pill. Title uppercase, accent color on keyword
  • Right — author, data source text + icon buttons, timestamp · all right-aligned, no-wrap
